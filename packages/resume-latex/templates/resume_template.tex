% ---------- FONT: Carlito (Calibri-metric-compatible) ----------------------
\usepackage{fontspec}
\setmainfont{Carlito}[Mapping=] % Mapping= (empty) disables fontspec's default
    % "tex-text" substitution, which otherwise silently converts straight
    % apostrophes/quotes to curly Unicode glyphs and "--"/"---" to en/em
    % dashes. This resume style bans smart quotes and en/em dashes outright
    % (ATS parsing risk) -- this line is what keeps the PDF text layer literal.
\renewcommand{\familydefault}{\rmdefault} % Carlito used as sole family (sans-serif design, no serif mixed in)

% ---------- SPACING / LIST CONTROL ------------------------------------------
\usepackage{enumitem}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt} % ALL vertical spacing is controlled explicitly via
    % \vspace calls in the custom commands below. Do NOT load the `parskip`
    % package -- it sets a nonzero default \parskip that silently stacks on
    % top of every explicit \vspace call here and is what causes 1-page
    % content to overflow to 2 pages. This line must stay 0pt.

% ---------- LETTER-SPACING for uppercase section headings ------------------
% fontspec + Carlito supports OpenType LetterSpace feature for CSS-like tracking.
\newfontfamily\headingfont{Carlito}[LetterSpace=3.0] % approximates 0.5px CSS tracking

% ---------- CUSTOM COMMANDS (typography primitives) -------------------------

\newcommand{\resumecontact}[1]{%

% --- Horizontal rule under header ---
% CSS: 1.3px solid #1a1a1a, margin 7px 0 9px
\newcommand{\resumehrule}{%
    \par\vspace{3pt}
    {\color{rulecolor}\hrule height 1.1pt}
    \vspace{4pt}
}

% --- Section heading (uppercase, bold, bottom border) ---
% CSS: font-size 10.3pt, weight 700, uppercase, letter-spacing 0.5px,
%      border-bottom 0.85pt solid rulecolor, padding-bottom ~2px,
%      margin-top 10px (0 for first heading), margin-bottom 5px
% TeX note: negative vspace after heading positions the rule under the line
% box; -4pt overlaps glyphs on Carlito — use -3.85pt with 10.8pt leading.
\newcommand{\sectionheading}[1]{%
    \par\vspace{3.2pt}
    {\headingfont\fontsize{10.3pt}{10.8pt}\selectfont\bfseries\MakeUppercase{#1}}\par
    \vspace{-3.85pt}
    {\color{rulecolor}\hrule height 0.85pt}
    \vspace{1.7pt}
}
% First section heading on the page must have NO top margin — use this variant
% immediately after \resumehrule for the first section only:
\newcommand{\sectionheadingfirst}[1]{%
    {\headingfont\fontsize{10.3pt}{10.8pt}\selectfont\bfseries\MakeUppercase{#1}}\par
    \vspace{-3.85pt}
    {\color{rulecolor}\hrule height 0.85pt}
    \vspace{5pt}
}

% --- Summary paragraph ---
% CSS: font-size 9.5pt, line-height 1.34, text-align justify
% NOTE: line-height is fully encoded in the \fontsize{9.5pt}{12.7pt} leading
% argument (12.7/9.5 ≈ 1.34) -- do not wrap this in a `setspace` \spacing
% environment, which would multiply the leading a second time.
\newcommand{\resumesummary}[1]{%
    {\fontsize{9.1pt}{10.7pt}\selectfont\color{textcolor} #1\par}
}

% --- Skills line (category label + comma list) ---
% CSS: font-size 9.2pt, line-height 1.5, margin-bottom 3px; label is bold
\newcommand{\skillsline}[2]{%
    \par\vspace{1.2pt}
    {\fontsize{8.9pt}{10.9pt}\selectfont\color{textcolor}\textbf{#1:} #2}
}
